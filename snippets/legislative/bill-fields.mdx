<ResponseField name="id" type="string">
  UUID of the bill.
</ResponseField>
<ResponseField name="jurisdiction" type="object">
  The legislature the bill belongs to, with a `code` (for example, `"GB"`) and a `name`. See [List jurisdictions](/api-reference/legislative/list-jurisdictions).
</ResponseField>
<ResponseField name="external_id" type="string">
  The bill's ID in the legislature's own system.
</ResponseField>
<ResponseField name="title" type="string">
  Short title of the bill.
</ResponseField>
<ResponseField name="long_title" type="string">
  Long title of the bill. Empty if unavailable.
</ResponseField>
<ResponseField name="bill_type" type="string">
  Bill subtype, such as `"Government Bill"` or `"Private Members' Bill (Presentation Bill)"`. Empty if unavailable.
</ResponseField>
<ResponseField name="session" type="string">
  The legislature's ID for the session, such as `"40"` or `"2025-2026"`.
</ResponseField>
<ResponseField name="session_name" type="string">
  Display name of the session, such as `"2026-27 session"`.
</ResponseField>
<ResponseField name="introduction_date" type="string | null">
  When the bill was introduced (ISO 8601).
</ResponseField>
<ResponseField name="current_house" type="string">
  The house currently considering the bill.
</ResponseField>
<ResponseField name="current_stage" type="string">
  The stage the bill is currently at, in the legislature's own language.
</ResponseField>
<ResponseField name="current_stage_date" type="string | null">
  When the bill's most recent step happened (ISO 8601). If the current stage has several sittings, this is the latest one so far. If the current stage hasn't started yet, this is the date of the bill's last step before it.
</ResponseField>
<ResponseField name="status_normalized" type="string">
  The bill's status, normalized across jurisdictions. One of `"INTRODUCED"`, `"CONSIDERING"`, `"PASSED"`, `"FAILED"`, `"INACTIVE"`, or `"OTHER"`. Empty if unknown.
</ResponseField>
<ResponseField name="originating_house" type="string">
  The house the bill was introduced in.
</ResponseField>
<ResponseField name="is_act" type="boolean">
  Whether the bill has become an act.
</ResponseField>
<ResponseField name="last_update" type="string">
  When the legislature last updated the bill (ISO 8601).
</ResponseField>
<ResponseField name="next_event_date" type="string | null">
  Date of the bill's next scheduled event (ISO 8601).
</ResponseField>
<ResponseField name="next_event_description" type="string">
  Description of the bill's next scheduled event.
</ResponseField>
<ResponseField name="amendments_as_at" type="string | null">
  When the bill's amendment list was last published (ISO 8601).
</ResponseField>
<ResponseField name="upstream_url" type="string">
  URL of the bill's page on the legislature's website.
</ResponseField>
<ResponseField name="source" type="string | null">
  `uuid` of the `"LEGISLATION"` source that holds the bill's current text. `null` if the current text hasn't been collected.
</ResponseField>
<ResponseField name="first_seen" type="string">
  When Parsagon first collected the bill (ISO 8601).
</ResponseField>
<ResponseField name="last_seen" type="string">
  When Parsagon last collected a change to the bill (ISO 8601). The `updated_since` and `updated_before` filters on [List bills](/api-reference/legislative/list-bills) use this field.
</ResponseField>
